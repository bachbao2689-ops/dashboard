3. DATABASE SCHEMA (COMPLETE - COPY TO MIGRATION)
3.1 Core Tables
-- =====================================================
-- WORKSPACES & USERS
-- =====================================================

CREATE TABLE users (
    id BIGSERIAL PRIMARY KEY,
    email VARCHAR(255) UNIQUE NOT NULL,
    name VARCHAR(200) NOT NULL,
    avatar_url TEXT,
    initials VARCHAR(5),  -- e.g., "L", "W" for display
    role VARCHAR(20) DEFAULT 'member',  -- admin, member, viewer
    department_id UUID,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    last_sign_in_at TIMESTAMPTZ
);

CREATE TABLE workspaces (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(100) NOT NULL,
    slug VARCHAR(50) UNIQUE NOT NULL,  -- "k-coffee"
    logo_url TEXT,
    tagline TEXT,  -- "ĐỨC SƠN NGUYÊN CỨU"
    owner_id BIGINT NOT NULL REFERENCES users(id),
    created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE workspace_members (
    workspace_id UUID REFERENCES workspaces(id) ON DELETE CASCADE,
    user_id BIGINT REFERENCES users(id) ON DELETE CASCADE,
    role VARCHAR(20) NOT NULL DEFAULT 'member',
    joined_at TIMESTAMPTZ DEFAULT NOW(),
    PRIMARY KEY (workspace_id, user_id)
);

-- =====================================================
-- DEPARTMENTS (From "Phòng ban" column in Tasks)
-- =====================================================

CREATE TABLE departments (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    workspace_id UUID NOT NULL REFERENCES workspaces(id) ON DELETE CASCADE,
    name VARCHAR(100) NOT NULL,  -- ECOMMERCE, KSHOP, MARKETING, DESIGN
    code VARCHAR(20),  -- Optional short code
    head_id BIGINT REFERENCES users(id),
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- =====================================================
-- TASKS MODULE
-- =====================================================

CREATE TABLE projects (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    workspace_id UUID NOT NULL REFERENCES workspaces(id) ON DELETE CASCADE,
    name VARCHAR(200) NOT NULL,  -- "ECOM BRAND REDESIGN", "Campaign tháng 8"
    description TEXT,
    status VARCHAR(50) DEFAULT 'active',  -- active, completed, on_hold
    start_date DATE,
    due_date DATE,
    created_by BIGINT NOT NULL REFERENCES users(id),
    created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE tasks (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    workspace_id UUID NOT NULL REFERENCES workspaces(id) ON DELETE CASCADE,
    project_id UUID REFERENCES projects(id) ON DELETE SET NULL,
    
    -- Task Identifiers
    task_ref VARCHAR(20) UNIQUE,  -- TK01, TK02 (auto-increment or custom)
    
    -- Core Fields (matching UI)
    title VARCHAR(500) NOT NULL,  -- "Update bộ hình sản phẩm"
    description TEXT,  -- Rich text or markdown
    
    -- Metadata (from UI columns)
    department_id UUID REFERENCES departments(id),
    assignee_id BIGINT REFERENCES users(id),
    
    -- Properties
    priority VARCHAR(20) DEFAULT 'medium',  -- low, medium, high
    status VARCHAR(50) DEFAULT 'todo',  -- todo, in_progress, feedback, done, pending
    
    -- Dates
    start_date DATE,
    due_date DATE,  -- "03/07/2026" format in UI
    completed_at TIMESTAMPTZ,
    
    -- Tracking
    attachments_count INTEGER DEFAULT 0,
    comments_count INTEGER DEFAULT 0,
    
    -- Kanban Board Position
    column_id UUID REFERENCES columns(id),  -- "yet_to_start", "in_progress"...
    position NUMERIC(20, 6) DEFAULT 0,  -- For drag-drop ordering
    
    -- Audit
    created_by BIGINT NOT NULL REFERENCES users(id),
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE INDEX idx_tasks_project ON tasks(project_id);
CREATE INDEX idx_tasks_assignee ON tasks(assignee_id) WHERE status != 'done';
CREATE INDEX idx_tasks_status ON tasks(status) WHERE status != 'done';
CREATE INDEX idx_tasks_due_date ON tasks(due_date) WHERE due_date IS NOT NULL;
CREATE INDEX idx_tasks_department ON tasks(department_id);

-- =====================================================
-- KANBAN COLUMNS (Fixed by default, extensible)
-- =====================================================

CREATE TABLE columns (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    workspace_id UUID NOT NULL REFERENCES workspaces(id) ON DELETE CASCADE,
    name VARCHAR(100) NOT NULL,  -- "Yet to Start", "In Progress"
    color VARCHAR(20),  -- gray, blue, orange, green
    position INTEGER NOT NULL,
    is_default BOOLEAN DEFAULT FALSE,
    UNIQUE(workspace_id, name)
);

-- =====================================================
-- ASSETS MODULE
-- =====================================================

CREATE TABLE asset_categories (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    workspace_id UUID NOT NULL REFERENCES workspaces(id) ON DELETE CASCADE,
    name VARCHAR(100) NOT NULL,  -- Laptop, Camera, Lens, Equipment
    icon VARCHAR(10),  -- emoji or icon key
    created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE assets (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    workspace_id UUID NOT NULL REFERENCES workspaces(id) ON DELETE CASCADE,
    
    -- Asset Identification (matching UI)
    asset_code VARCHAR(50) UNIQUE NOT NULL,  -- LENCANON01, MAYANHCANON01
    name VARCHAR(255) NOT NULL,  -- "Lens Canon 24-70mm Ultrasonic"
    
    -- Categorization
    category_id UUID REFERENCES asset_categories(id),
    
    -- Properties (from UI)
    condition VARCHAR(50),  -- "Tốt 100%", "N/A"
    location VARCHAR(200),  -- "Kho Công Ty"
    
    -- Status
    status VARCHAR(50) DEFAULT 'available',  -- available, borrowed, maintenance
    is_available BOOLEAN DEFAULT TRUE,
    
    -- Details
    description TEXT,
    purchase_date DATE,
    purchase_price DECIMAL(15, 2),
    serial_number VARCHAR(100),
    
    -- Images
    image_url TEXT,
    
    -- Maintenance
    maintenance_scheduled DATE,
    maintenance_notes TEXT,
    
    -- Audit
    added_by BIGINT NOT NULL REFERENCES users(id),
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE INDEX idx_assets_code ON assets(asset_code);
CREATE INDEX idx_assets_category ON assets(category_id);
CREATE INDEX idx_assets_status ON assets(status);
CREATE INDEX idx_assets_availability ON assets(is_available);

-- =====================================================
-- BORROW RECORDS (From "Nhật ký mượn tài sản")
-- =====================================================

CREATE TABLE borrow_requests (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    workspace_id UUID NOT NULL REFERENCES workspaces(id) ON DELETE CASCADE,
    
    -- Asset & Borrower
    asset_id UUID NOT NULL REFERENCES assets(id) ON DELETE CASCADE,
    requester_id BIGINT NOT NULL REFERENCES users(id),
    department_id UUID REFERENCES departments(id),
    
    -- Dates
    requested_at TIMESTAMPTZ DEFAULT NOW(),
    borrow_date DATE NOT NULL,  -- "Ngày mượn"
    due_date DATE NOT NULL,  -- "Hạn trả"
    returned_at TIMESTAMPTZ,
    
    -- Purpose & Condition
    purpose TEXT,  -- "Chụp ngoại cảnh set quà 20.10"
    condition_at_borrow TEXT,  -- "Nguyên 100%"
    
    -- Approval Workflow
    approval_status VARCHAR(50) DEFAULT 'pending',  -- pending, approved, rejected
    approved_by BIGINT REFERENCES users(id),
    approved_at TIMESTAMPTZ,
    
    -- Return Confirmation
    return_condition TEXT,  -- "Tình trạng" khi trả
    return_confirmed_by BIGINT REFERENCES users(id),
    return_confirmed_at TIMESTAMPTZ,
    
    -- Notes
    notes TEXT,
    
    created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE INDEX idx_borrow_asset ON borrow_requests(asset_id);
CREATE INDEX idx_borrow_requester ON borrow_requests(requester_id);
CREATE INDEX idx_borrow_pending ON borrow_requests(approval_status) 
    WHERE approval_status = 'pending';
CREATE INDEX idx_borrow_active ON borrow_requests(asset_id)
    WHERE returned_at IS NULL AND approval_status = 'approved';

-- =====================================================
-- COMMENTS & ACTIVITY
-- =====================================================

CREATE TABLE comments (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    task_id UUID NOT NULL REFERENCES tasks(id) ON DELETE CASCADE,
    author_id BIGINT NOT NULL REFERENCES users(id),
    body TEXT NOT NULL,
    mentions BIGINT[] DEFAULT '{}',
    is_edited BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE INDEX idx_comments_task ON comments(task_id);

CREATE TABLE activity_log (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    workspace_id UUID NOT NULL REFERENCES workspaces(id) ON DELETE CASCADE,
    user_id BIGINT NOT NULL REFERENCES users(id),
    action VARCHAR(50) NOT NULL,  -- 'task_created', 'status_changed', 'asset_borrowed'
    entity_type VARCHAR(50),  -- 'task', 'asset', 'borrow_request'
    entity_id UUID,
    old_value JSONB,
    new_value JSONB,
    metadata JSONB DEFAULT '{}',
    created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE INDEX idx_activity_workspace ON activity_log(workspace_id, created_at DESC);

-- =====================================================
-- NOTIFICATIONS
-- =====================================================

CREATE TABLE notifications (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id BIGINT NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    type VARCHAR(50) NOT NULL,  -- 'task_assigned', 'borrow_approved', 'overdue'
    entity_type VARCHAR(50),  -- 'task', 'asset', 'borrow_request'
    entity_id UUID,
    message TEXT NOT NULL,
    metadata JSONB DEFAULT '{}',
    is_read BOOLEAN DEFAULT FALSE,
    read_at TIMESTAMPTZ,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE INDEX idx_notifications_unread ON notifications(user_id) WHERE is_read = FALSE;

-- =====================================================
-- TRIGGERS & HELPER FUNCTIONS
-- =====================================================

-- Auto-update updated_at
CREATE OR REPLACE FUNCTION update_updated_at_column()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = NOW();
    RETURN NEW;
END;
$$ language 'plpgsql';

CREATE TRIGGER update_tasks_updated_at BEFORE UPDATE ON tasks
    FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();

CREATE TRIGGER update_assets_updated_at BEFORE UPDATE ON assets
    FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();

-- Update task status when moved to "Completed" column
CREATE OR REPLACE FUNCTION update_task_status_on_column_change()
RETURNS TRIGGER AS $$
BEGIN
    IF NEW.column_id != OLD.column_id THEN
        IF NEW.column_id = (SELECT id FROM columns WHERE workspace_id = NEW.workspace_id AND name = 'Completed') THEN
            NEW.status := 'done';
            NEW.completed_at := NOW();
        ELSIF NEW.column_id = (SELECT id FROM columns WHERE workspace_id = NEW.workspace_id AND name = 'Feedback') THEN
            NEW.status := 'feedback';
        ELSIF NEW.column_id = (SELECT id FROM columns WHERE workspace_id = NEW.workspace_id AND name = 'In Progress') THEN
            NEW.status := 'in_progress';
        ELSE
            NEW.status := 'todo';
        END IF;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER task_update_status_trigger 
    BEFORE UPDATE ON tasks
    FOR EACH ROW EXECUTE FUNCTION update_task_status_on_column_change();

// /server/services/taskService.ts

interface TaskFilters {
  departmentId?: string;
  assigneeId?: number;
  status?: string[];
  priority?: string[];
  projectId?: string;
  dateRange?: { start: string; end: string };
  search?: string;
}

async function getTasks(
  workspaceId: string,
  userId: number,
  filters: TaskFilters,
  pagination: { page: number; limit: number },
  sorting: { sortBy: string; sortOrder: 'asc' | 'desc' }
): Promise<TaskQueryResult> {
  
  // Build query with permission checks
  const baseQuery = db.tasks
    .where('workspace_id', workspaceId)
    .leftJoin('projects', 'tasks.project_id', 'projects.id')
    .leftJoin('departments', 'tasks.department_id', 'departments.id')
    .leftJoin('users', 'tasks.assignee_id', 'users.id');
  
  // Apply filters
  if (filters.departmentId) {
    baseQuery.andWhere('tasks.department_id', filters.departmentId);
  }
  
  if (filters.status && filters.status.length > 0) {
    baseQuery.andIn('tasks.status', filters.status);
  }
  
  if (filters.assigneeId === userId) {
    // User sees their own tasks + team tasks based on role
    if (userRole !== 'admin') {
      baseQuery.andWhere('tasks.assignee_id', userId);
    }
  }
  
  // Search
  if (filters.search) {
    baseQuery.andWhere((q) =>
      q.whereRaw('tasks.title ILIKE ?', [`%${filters.search}%`])
        .orWhereRaw('tasks.description ILIKE ?', [`%${filters.search}%`])
    );
  }
  
  // Sorting
  const orderDirection = sorting.sortOrder.toUpperCase();
  baseQuery.orderBy(sorting.sortBy, orderDirection);
  
  // Pagination
  const offset = (pagination.page - 1) * pagination.limit;
  const tasks = await baseQuery
    .limit(pagination.limit)
    .offset(offset);
  
  const total = await baseQuery.count('* as count').first();
  
  return {
    data: tasks,
    meta: {
      totalCount: parseInt(total.count),
      currentPage: pagination.page,
      totalPages: Math.ceil(parseInt(total.count) / pagination.limit),
      hasMore: tasks.length === pagination.limit
    }
  };
}

async function createTask(workspaceId: string, userId: number, data: CreateTaskDto): Promise<Task> {
  // Validate permissions
  await checkPermission(workspaceId, userId, 'CREATE_TASK');
  
  // Generate task_ref if not provided (TK01, TK02...)
  const latestTask = await db.tasks
    .where('workspace_id', workspaceId)
    .orderBy('created_at', 'desc')
    .first();
  
  const nextNum = latestTask 
    ? parseInt(latestTask.task_ref.replace('TK', '')) + 1 
    : 1;
  const taskRef = `TK${nextNum.toString().padStart(2, '0')}`;
  
  // Insert
  const [task] = await db.tasks.insert({
    workspace_id: workspaceId,
    task_ref: taskRef,
    title: data.title,
    description: data.description,
    department_id: data.departmentId,
    assignee_id: data.assigneeId,
    priority: data.priority || 'medium',
    status: data.status || 'todo',
    due_date: data.dueDate,
    start_date: data.startDate,
    project_id: data.projectId,
    column_id: data.columnId,
    created_by: userId
  }).returning('*');
  
  // Log activity
  await db.activityLog.insert({
    workspace_id: workspaceId,
    user_id: userId,
    action: 'task_created',
    entity_type: 'task',
    entity_id: task.id,
    new_value: { title: task.title, status: task.status }
  });
  
  // Notify assignee if assigned
  if (data.assigneeId && data.assigneeId !== userId) {
    await notificationService.notify(data.assigneeId, {
      type: 'task_assigned',
      message: `You have been assigned to task ${taskRef}: ${task.title}`,
      entityType: 'task',
      entityId: task.id
    });
  }
  
  return task;
}

async function updateTaskStatus(taskId: string, newStatus: string, userId: number): Promise<Task> {
  // Get current task
  const task = await db.tasks.findById(taskId);
  
  if (!task) throw new NotFoundError('Task not found');
  
  // Check permissions
  await checkPermission(task.workspace_id, userId, 'UPDATE_TASK');
  
  // Update
  const [updated] = await db.tasks
    .update({
      status: newStatus,
      updated_at: NOW(),
      ...(newStatus === 'done' ? { completed_at: NOW() } : {})
    })
    .where('id', taskId)
    .returning('*');
  
  // Log
  await db.activityLog.insert({
    workspace_id: task.workspace_id,
    user_id: userId,
    action: 'status_changed',
    entity_type: 'task',
    entity_id: taskId,
    old_value: { status: task.status },
    new_value: { status: newStatus }
  });
  
  return updated;
}

async function moveTaskInKanban(taskId: string, newColumnId: string, newPosition: number, userId: number): Promise<Task> {
  const task = await db.tasks.findById(taskId);
  
  if (!task) throw new NotFoundError('Task not found');
  
  // Get column to determine new status
  const column = await db.columns.findById(newColumnId);
  const newStatus = mapColumnToStatus(column.name);  // "In Progress" → "in_progress"
  
  // Update
  const [updated] = await db.tasks
    .update({
      column_id: newColumnId,
      position: newPosition,
      status: newStatus,
      ...(newStatus === 'done' ? { completed_at: NOW() } : {}),
      updated_at: NOW()
    })
    .where('id', taskId)
    .returning('*');
  
  // Log
  await db.activityLog.insert({
    workspace_id: task.workspace_id,
    user_id: userId,
    action: 'task_moved',
    entity_type: 'task',
    entity_id: taskId,
    old_value: { columnId: task.column_id, status: task.status },
    new_value: { columnId: newColumnId, status: newStatus, position: newPosition }
  });
  
  return updated;
}

async function getAssets(
  workspaceId: string,
  filters: AssetFilters
): Promise<AssetQueryResult> {
  
  let query = db.assets
    .where('workspace_id', workspaceId)
    .leftJoin('asset_categories', 'assets.category_id', 'asset_categories.id');
  
  // Filters
  if (filters.search) {
    query.andWhere((q) =>
      q.whereRaw('assets.asset_code ILIKE ?', [`%${filters.search}%`])
        .orWhereRaw('assets.name ILIKE ?', [`%${filters.search}%`])
    );
  }
  
  if (filters.status) {
    query.andWhere('assets.status', filters.status);
  }
  
  if (filters.categoryId) {
    query.andWhere('assets.category_id', filters.categoryId);
  }
  
  const assets = await query.select([
    'assets.*',
    'asset_categories.name as category_name'
  ]);
  
  return {
    data: assets,
    meta: { totalCount: assets.length }
  };
}

async function createAsset(workspaceId: string, userId: number, data: CreateAssetDto): Promise<Asset> {
  await checkPermission(workspaceId, userId, 'MANAGE_ASSETS');
  
  // Validate asset code uniqueness
  const existing = await db.assets.findOne({ asset_code: data.assetCode });
  if (existing) {
    throw new ValidationError('Asset code already exists');
  }
  
  const [asset] = await db.assets.insert({
    workspace_id: workspaceId,
    asset_code: data.assetCode,
    name: data.name,
    category_id: data.categoryId,
    condition: data.condition,
    location: data.location,
    status: 'available',
    is_available: true,
    description: data.description,
    purchase_date: data.purchaseDate,
    purchase_price: data.purchasePrice,
    serial_number: data.serialNumber,
    image_url: data.imageUrl,
    added_by: userId
  }).returning('*');
  
  await db.activityLog.insert({
    workspace_id: workspaceId,
    user_id: userId,
    action: 'asset_created',
    entity_type: 'asset',
    entity_id: asset.id,
    new_value: { name: asset.name, code: asset.asset_code }
  });
  
  return asset;
}

async function approveBorrowRequest(requestId: string, approverId: number, notes?: string): Promise<BorrowRequest> {
  const request = await db.borrowRequests.findById(requestId);
  
  if (!request) throw new NotFoundError('Borrow request not found');
  
  // Check if asset is available
  const asset = await db.assets.findById(request.asset_id);
  if (!asset.is_available) {
    throw new ValidationError('Asset is not available for borrowing');
  }
  
  // Check for overlapping borrow dates
  const conflicting = await db.borrowRequests.findOne({
    asset_id: request.asset_id,
    approval_status: 'approved',
    borrow_date: { lte: request.due_date },
    due_date: { gte: request.borrow_date },
    returned_at: null
  });
  
  if (conflicting) {
    throw new ValidationError('Asset is already borrowed during these dates');
  }
  
  // Approve
  const [updated] = await db.borrowRequests
    .update({
      approval_status: 'approved',
      approved_by: approverId,
      approved_at: NOW()
    })
    .where('id', requestId)
    .returning('*');
  
  // Mark asset as borrowed
  await db.assets.update({
    status: 'borrowed',
    is_available: false
  }).where('id', request.asset_id);
  
  // Notify requester
  await notificationService.notify(request.requester_id, {
    type: 'borrow_approved',
    message: `Your borrow request for ${asset.name} has been approved`,
    entityType: 'borrow_request',
    entityId: requestId
  });
  
  return updated;
}

async function rejectBorrowRequest(requestId: string, rejecterId: number, reason: string): Promise<BorrowRequest> {
  const [updated] = await db.borrowRequests
    .update({
      approval_status: 'rejected',
      approved_by: rejecterId,
      approved_at: NOW()
    })
    .where('id', requestId)
    .returning('*');
  
  // Notify requester
  await notificationService.notify(updated.requester_id, {
    type: 'borrow_rejected',
    message: `Your borrow request was rejected: ${reason}`,
    entityType: 'borrow_request',
    entityId: requestId
  });
  
  return updated;
}

async function getDashboardData(workspaceId: string, userId: number, timeRange: string): Promise<DashboardData> {
  const user = await db.users.findById(userId);
  
  // KPIs
  const myTasksCount = await db.tasks.count({
    workspace_id: workspaceId,
    assignee_id: userId,
    status: { neq: 'done' }
  });
  
  const dueSoonCount = await db.tasks.count({
    workspace_id: workspaceId,
    due_date: { gte: today, lte: nextWeek },
    status: { neq: 'done' }
  });
  
  const borrowedCount = await db.borrowRequests.count({
    workspace_id: workspaceId,
    requester_id: userId,
    returned_at: null,
    approval_status: 'approved'
  });
  
  const overdueCount = await db.tasks.count({
    workspace_id: workspaceId,
    due_date: { lt: today },
    status: { neq: 'done' }
  });
  
  // Task Status Chart (weekly)
  const taskStatusData = await db.raw(`
    SELECT 
      EXTRACT(WEEK FROM created_at) as week,
      COUNT(*) FILTER (WHERE status = 'done' OR completed_at IS NOT NULL) as done_count,
      COUNT(*) FILTER (WHERE status != 'done') as remaining_count
    FROM tasks
    WHERE workspace_id = $1 AND created_at >= (NOW() - INTERVAL '30 days')
    GROUP BY week
    ORDER BY week
  `, [workspaceId]);
  
  // Asset Utilization
  const assetStats = await db.raw(`
    SELECT
      COUNT(*) FILTER (WHERE status = 'borrowed') as borrowed,
      COUNT(*) FILTER (WHERE status = 'available') as available,
      COUNT(*) FILTER (WHERE status = 'maintenance') as maintenance,
      COUNT(*) as total
    FROM assets
    WHERE workspace_id = $1
  `, [workspaceId]);
  
  // Upcoming Deadlines
  const upcomingDeadlines = await db.tasks
    .where('workspace_id', workspaceId)
    .andWhere('due_date', '>=', today)
    .andWhere('due_date', '<=', nextWeek)
    .andWhere('status', '!=', 'done')
    .orderBy('due_date', 'asc')
    .limit(10)
    .leftJoin('users', 'tasks.assignee_id', 'users.id')
    .select([
      'tasks.*',
      'users.name as assignee_name',
      'users.initials as assignee_initials'
    ]);
  
  // Recent Borrow Requests (pending)
  const recentBorrowRequests = await db.borrowRequests
    .where('workspace_id', workspaceId)
    .andWhere('approval_status', 'pending')
    .orderBy('requested_at', 'desc')
    .limit(5)
    .leftJoin('assets', 'borrow_requests.asset_id', 'assets.id')
    .leftJoin('users', 'borrow_requests.requester_id', 'users.id')
    .leftJoin('departments', 'borrow_requests.department_id', 'departments.id')
    .select([
      'borrow_requests.*',
      'assets.name as asset_name',
      'users.name as requester_name',
      'departments.name as department_name'
    ]);
  
  return {
    welcomeMessage: `Welcome, ${user.name} 👋`,
    kpis: [
      { id: 'my_tasks', title: 'My Tasks', value: myTasksCount, delta: '+5%', direction: 'up', color: 'primary' },
      { id: 'due_soon', title: 'Due Soon', value: dueSoonCount, delta: '-2%', direction: 'down', color: 'warning' },
      { id: 'borrowed', title: 'Borrowed', value: borrowedCount, delta: '+1%', direction: 'up', color: 'info' },
      { id: 'overdue', title: 'Overdue', value: overdueCount, color: 'danger', badge: 'Action Required' }
    ],
    charts: [
      {
        id: 'task_status_this_month',
        title: 'Task Status This Month',
        type: 'line_chart',
        data: {
          weeks: taskStatusData.map(r => `Week ${r.week}`),
          series: [
            { name: 'Created', color: '#1e3a8a', values: taskStatusData.map(r => r.remaining_count) },
            { name: 'Done', color: '#14b8a6', values: taskStatusData.map(r => r.done_count) }
          ]
        }
      },
      {
        id: 'asset_utilization',
        title: 'Asset Utilization',
        type: 'donut_chart',
        data: {
          segments: [
            { label: 'Borrowed', value: Math.round((assetStats[0].borrowed / assetStats[0].total) * 100), color: '#1e3a8a' },
            { label: 'Available', value: Math.round((assetStats[0].available / assetStats[0].total) * 100), color: '#14b8a6' },
            { label: 'Maintenance', value: Math.round((assetStats[0].maintenance / assetStats[0].total) * 100), color: '#ef4444' }
          ]
        }
      }
    ],
    lists: {
      upcomingDeadlines: upcomingDeadlines.map(t => ({
        id: t.task_ref,
        title: t.title,
        priority: t.priority,
        dueDate: formatDate(t.due_date),
        assignee: { name: t.assignee_name || 'Unassigned', initials: t.assignee_initials }
      })),
      recentBorrowRequests: recentBorrowRequests.map(r => ({
        id: r.id,
        requesterName: r.requester_name,
        assetName: r.asset_name,
        dueDate: formatDate(r.due_date),
        status: r.approval_status
      }))
    }
  };
}

6.2 State Management Setup
// /stores/dashboardStore.ts
import create from 'zustand';

interface DashboardState {
  data: DashboardData | null;
  isLoading: boolean;
  error: string | null;
  lastUpdated: Date | null;
  fetchData: (workspaceId: string) => Promise<void>;
  refresh: () => void;
}

export const useDashboardStore = create<DashboardState>((set, get) => ({
  data: null,
  isLoading: false,
  error: null,
  lastUpdated: null,
  
  fetchData: async (workspaceId: string) => {
    set({ isLoading: true, error: null });
    try {
      const res = await api.get(`/workspaces/${workspaceId}/dashboard`);
      set({ data: res.data, isLoading: false, lastUpdated: new Date() });
    } catch (err) {
      set({ error: err.message, isLoading: false });
    }
  },
  
  refresh: () => {
    const { fetchData } = get();
    fetchData(get().currentWorkspaceId);
  }
}));

// /components/Home/index.tsx
export default function Home() {
  const { data, isLoading, error, refresh } = useDashboardStore();
  
  useEffect(() => {
    refresh();
    // Auto-refresh every 60s
    const interval = setInterval(refresh, 60000);
    return () => clearInterval(interval);
  }, []);
  
  if (isLoading) return <DashboardSkeleton />;
  if (error) return <ErrorState message={error} />;
  
  return <DashboardContent data={data} onRefresh={refresh} />;
}
