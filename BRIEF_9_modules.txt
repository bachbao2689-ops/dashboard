3.4.2 Urgency Logic
const urgencyLevels = {
  danger: { daysOverdue: 1, color: 'orange', bg: '#FEF3E2' },
  warning: { daysOverdue: 3, color: 'red', bg: '#FEE2E2' },
  critical: { daysOverdue: 7, color: 'darkred', bg: '#FE2A2A', text: '#FFFFFF' }
};

function calculateUrgency(task) {
  const daysLate = Math.floor((today - task.dueDate) / 86400000);
  return Object.values(urgencyLevels).find(l => daysLate >= l.daysOverdue);
}

3.7.2 Calculation Logic
function calculateWorkload(member) {
  const activeTasks = tasks.filter(t => t.assignee === member.id && t.status !== 'done');
  const totalEstimate = activeTasks.reduce((sum, t) => sum + t.timeEstimate, 0);
  const weeklyCapacity = 40; // hours
  
  return {
    percentage: (totalEstimate / weeklyCapacity) * 100,
    status: totalEstimate > weeklyCapacity ? 'OVERLOADED' :
            totalEstimate > weeklyCapacity * 0.8 ? 'NEAR_LIMIT' :
            'BALANCED'
  };
}
