3. Combined Data Model
3.1 Unified Schema
-- DEPARTMENTS (from "Phòng ban" column)
CREATE TABLE departments (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(100) NOT NULL,  -- PS COFFEE, MARKETING, ECOMMERCE...
    code VARCHAR(20),  -- Optional short code
    head_id BIGINT REFERENCES users(id),
    created_at TIMESTAMP DEFAULT NOW()
);

-- CHANNELS/PLATFORMS (from "Nền tảng" column)
CREATE TABLE channels (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(50) NOT NULL,  -- SYSTEM, Zalo, Events, Facebook, TikTok...
    category VARCHAR(50),  -- SOCIAL, ECOMMERCE, OFFLINE...
    icon_url TEXT
);

-- TASKS (from Task Sheet, enhanced with relations)
CREATE TABLE tasks (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    task_ref VARCHAR(20) UNIQUE,  -- TK75, TK81 (migratable from Google Sheet)
    
    -- Core fields
    department_id UUID NOT NULL REFERENCES departments(id),
    channel_ids UUID[] NOT NULL,  -- Array of channel refs (multi-select)
    project_name VARCHAR(200) NOT NULL,  -- "DỰ ÁN" column
    title VARCHAR(255) NOT NULL,  -- "Tên công việc"
    description TEXT,  -- "Nội dung"
    
    -- Assignments
    pic_id BIGINT NOT NULL REFERENCES users(id),  -- "Người phụ trách"
    
    -- Priorities & Status
    priority VARCHAR(20) NOT NULL,  -- low, medium, high, urgent
    status VARCHAR(50) NOT NULL,  -- todo, in_progress, blocked, done, feedback
    
    -- Dates (all nullable, depending on task type)
    start_date DATE,
    due_date DATE,  -- "Ngày kết thúc"
    air_date DATE,  -- "Ngày đăng bài" (content publishing)
    
    -- Deliverables
    deliverable_desc TEXT,  -- "Sản phẩm cần giao"
    deliverable_files UUID[] REFERENCES files(id),  -- Upload attachments
    
    -- Workflow metadata
    parent_task_id UUID REFERENCES tasks(id),  -- Subtasks support
    tags TEXT[],  -- Custom tags
    
    -- Audit
    created_by BIGINT NOT NULL REFERENCES users(id),
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP DEFAULT NOW(),
    completed_at TIMESTAMP
);

CREATE INDEX idx_tasks_dept_status ON tasks(department_id, status) WHERE status != 'done';
CREATE INDEX idx_tasks_pic_active ON tasks(pic_id, status) WHERE status NOT IN ('done', 'cancelled');
CREATE INDEX idx_tasks_due_date ON tasks(due_date) WHERE due_date IS NOT NULL;

-- ASSETS (from Asset Sheet)
CREATE TABLE assets (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    asset_code VARCHAR(50) UNIQUE NOT NULL,  -- "Mã Thiết bị"
    name VARCHAR(200) NOT NULL,  -- "Tên Thiết bị / Model"
    category VARCHAR(100),  -- Camera, Laptop, Lens...
    
    -- Condition tracking
    current_condition VARCHAR(50),  -- "Tình trạng" (Original 100%)
    location VARCHAR(200),  -- Where is it physically stored
    
    -- Availability
    is_available BOOLEAN DEFAULT TRUE,
    maintenance_scheduled DATE,
    
    -- Photos/docs
    image_url TEXT,
    manual_url TEXT,  -- PDF link
    
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP DEFAULT NOW()
);

CREATE INDEX idx_assets_category ON assets(category);
CREATE INDEX idx_assets_availability ON assets(is_available);

-- BORROW RECORDS (from "Nhật ký quản lý mượn")
CREATE TABLE borrow_records (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    asset_id UUID NOT NULL REFERENCES assets(id),
    borrower_id BIGINT NOT NULL REFERENCES users(id),
    department_id UUID NOT NULL REFERENCES departments(id),
    
    -- Dates
    requested_at TIMESTAMP DEFAULT NOW(),
    borrow_date DATE NOT NULL,  -- "Ngày mượn"
    due_date DATE NOT NULL,  -- "Hạn trả"
    returned_at TIMESTAMP,
    
    -- Purpose & Condition
    purpose TEXT,  -- "Mục đích"
    condition_at_borrow TEXT,  -- "Tình trạng" khi mượn
    
    -- Approval workflow
    approval_status VARCHAR(50) DEFAULT 'pending',  -- pending, approved, rejected
    approved_by BIGINT REFERENCES users(id),
    approved_at TIMESTAMP,
    
    -- Return confirmation
    return_condition TEXT,  -- "Tình trạng" khi trả
    return_confirmed_by BIGINT REFERENCES users(id),
    return_confirmed_at TIMESTAMP,
    
    -- Notes
    notes TEXT,
    
    created_at TIMESTAMP DEFAULT NOW()
);

CREATE INDEX idx_borrow_asset_active ON borrow_records(asset_id) 
    WHERE returned_at IS NULL AND approval_status = 'approved';
CREATE INDEX idx_borrow_user_active ON borrow_records(borrower_id) 
    WHERE returned_at IS NULL;

-- WORKLOAD EVENTS (for adaptive layout)
CREATE TABLE workload_events (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id BIGINT NOT NULL,
    department_id UUID NOT NULL,
    task_count INTEGER DEFAULT 0,
    asset_borrow_count INTEGER DEFAULT 0,
    blocked_task_count INTEGER DEFAULT 0,
    overdue_task_count INTEGER DEFAULT 0,
    capacity_percentage INTEGER,  -- 0-100%
    recorded_at TIMESTAMP DEFAULT NOW()
);

-- NOTIFICATIONS (expanded for both task + asset)
CREATE TABLE notifications (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id BIGINT NOT NULL,
    type VARCHAR(50),  -- 'task_assigned', 'task_overdue', 'asset_request', 'asset_due_return'
    related_entity_type VARCHAR(50),  -- 'task' or 'asset'
    related_entity_id UUID,
    message TEXT,
    is_read BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT NOW()
);

CREATE INDEX idx_notifications_user_unread ON notifications(user_id, is_read) WHERE is_read = FALSE;

5.2 Row-Level Security Enforcement
// Middleware example
const applyRowLevelSecurity = async (query, user, role, context) => {
  if (role === 'viewer' || role === 'member') {
    // Only see own department
    query.where('department_id', '=', user.department_id);
  }
  
  if (role === 'member') {
    // Can't see other members' workload
    query.exclude('workload_report');
  }
  
  return query;
};

7. Migration Strategy (From Google Sheets)
7.1 Pre-Migration Checklist
[ ] Export both Google Sheets to CSV
[ ] Map columns: Spreadsheet → Database schema
[ ] Identify duplicates (same task ID, same asset code)
[ ] Clean up malformed data (invalid dates, missing required fields)
[ ] Define default values for NULL columns
[ ] Create staging environment with test data
7.2 Migration Script Flow
# migrate_from_sheets.py

# 1. Load CSVs
tasks_csv = pd.read_csv('manage_task_sheet.csv')
assets_csv = pd.read_csv('manage_asset_sheet.csv')

# 2. Transform
tasks_transformed = tasks_csv.rename(columns={
    'Phòng ban': 'department_name',
    'Nền tảng': 'channel_names',
    'Người phụ trách': 'pic_name'
})

# Resolve names to IDs
tasks_transformed['department_id'] = resolve_department_ids(tasks_transformed['department_name'])
tasks_transformed['pic_id'] = resolve_user_ids(tasks_transformed['pic_name'])

# 3. Validate
assert tasks_transformed['due_date'].apply(validate_date).all()
assert tasks_transformed['priority'].isin(['low','medium','high','urgent']).all()

# 4. Insert
batch_insert(tasks_transformed, batch_size=100)

# 5. Verify counts
assert db.count('tasks') == len(tasks_csv)
assert db.count('assets') == len(assets_csv)

3.1 Unified Schema
-- DEPARTMENTS (from "Phòng ban" column)
CREATE TABLE departments (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(100) NOT NULL,  -- PS COFFEE, MARKETING, ECOMMERCE...
    code VARCHAR(20),  -- Optional short code
    head_id BIGINT REFERENCES users(id),
    created_at TIMESTAMP DEFAULT NOW()
);

-- CHANNELS/PLATFORMS (from "Nền tảng" column)
CREATE TABLE channels (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(50) NOT NULL,  -- SYSTEM, Zalo, Events, Facebook, TikTok...
    category VARCHAR(50),  -- SOCIAL, ECOMMERCE, OFFLINE...
    icon_url TEXT
);

-- TASKS (from Task Sheet, enhanced with relations)
CREATE TABLE tasks (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    task_ref VARCHAR(20) UNIQUE,  -- TK75, TK81 (migratable from Google Sheet)
    
    -- Core fields
    department_id UUID NOT NULL REFERENCES departments(id),
    channel_ids UUID[] NOT NULL,  -- Array of channel refs (multi-select)
    project_name VARCHAR(200) NOT NULL,  -- "DỰ ÁN" column
    title VARCHAR(255) NOT NULL,  -- "Tên công việc"
    description TEXT,  -- "Nội dung"
    
    -- Assignments
    pic_id BIGINT NOT NULL REFERENCES users(id),  -- "Người phụ trách"
    
    -- Priorities & Status
    priority VARCHAR(20) NOT NULL,  -- low, medium, high, urgent
    status VARCHAR(50) NOT NULL,  -- todo, in_progress, blocked, done, feedback
    
    -- Dates (all nullable, depending on task type)
    start_date DATE,
    due_date DATE,  -- "Ngày kết thúc"
    air_date DATE,  -- "Ngày đăng bài" (content publishing)
    
    -- Deliverables
    deliverable_desc TEXT,  -- "Sản phẩm cần giao"
    deliverable_files UUID[] REFERENCES files(id),  -- Upload attachments
    
    -- Workflow metadata
    parent_task_id UUID REFERENCES tasks(id),  -- Subtasks support
    tags TEXT[],  -- Custom tags
    
    -- Audit
    created_by BIGINT NOT NULL REFERENCES users(id),
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP DEFAULT NOW(),
    completed_at TIMESTAMP
);

CREATE INDEX idx_tasks_dept_status ON tasks(department_id, status) WHERE status != 'done';
CREATE INDEX idx_tasks_pic_active ON tasks(pic_id, status) WHERE status NOT IN ('done', 'cancelled');
CREATE INDEX idx_tasks_due_date ON tasks(due_date) WHERE due_date IS NOT NULL;

-- ASSETS (from Asset Sheet)
CREATE TABLE assets (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    asset_code VARCHAR(50) UNIQUE NOT NULL,  -- "Mã Thiết bị"
    name VARCHAR(200) NOT NULL,  -- "Tên Thiết bị / Model"
    category VARCHAR(100),  -- Camera, Laptop, Lens...
    
    -- Condition tracking
    current_condition VARCHAR(50),  -- "Tình trạng" (Original 100%)
    location VARCHAR(200),  -- Where is it physically stored
    
    -- Availability
    is_available BOOLEAN DEFAULT TRUE,
    maintenance_scheduled DATE,
    
    -- Photos/docs
    image_url TEXT,
    manual_url TEXT,  -- PDF link
    
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP DEFAULT NOW()
);

CREATE INDEX idx_assets_category ON assets(category);
CREATE INDEX idx_assets_availability ON assets(is_available);

-- BORROW RECORDS (from "Nhật ký quản lý mượn")
CREATE TABLE borrow_records (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    asset_id UUID NOT NULL REFERENCES assets(id),
    borrower_id BIGINT NOT NULL REFERENCES users(id),
    department_id UUID NOT NULL REFERENCES departments(id),
    
    -- Dates
    requested_at TIMESTAMP DEFAULT NOW(),
    borrow_date DATE NOT NULL,  -- "Ngày mượn"
    due_date DATE NOT NULL,  -- "Hạn trả"
    returned_at TIMESTAMP,
    
    -- Purpose & Condition
    purpose TEXT,  -- "Mục đích"
    condition_at_borrow TEXT,  -- "Tình trạng" khi mượn
    
    -- Approval workflow
    approval_status VARCHAR(50) DEFAULT 'pending',  -- pending, approved, rejected
    approved_by BIGINT REFERENCES users(id),
    approved_at TIMESTAMP,
    
    -- Return confirmation
    return_condition TEXT,  -- "Tình trạng" khi trả
    return_confirmed_by BIGINT REFERENCES users(id),
    return_confirmed_at TIMESTAMP,
    
    -- Notes
    notes TEXT,
    
    created_at TIMESTAMP DEFAULT NOW()
);

CREATE INDEX idx_borrow_asset_active ON borrow_records(asset_id) 
    WHERE returned_at IS NULL AND approval_status = 'approved';
CREATE INDEX idx_borrow_user_active ON borrow_records(borrower_id) 
    WHERE returned_at IS NULL;

-- WORKLOAD EVENTS (for adaptive layout)
CREATE TABLE workload_events (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id BIGINT NOT NULL,
    department_id UUID NOT NULL,
    task_count INTEGER DEFAULT 0,
    asset_borrow_count INTEGER DEFAULT 0,
    blocked_task_count INTEGER DEFAULT 0,
    overdue_task_count INTEGER DEFAULT 0,
    capacity_percentage INTEGER,  -- 0-100%
    recorded_at TIMESTAMP DEFAULT NOW()
);

-- NOTIFICATIONS (expanded for both task + asset)
CREATE TABLE notifications (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id BIGINT NOT NULL,
    type VARCHAR(50),  -- 'task_assigned', 'task_overdue', 'asset_request', 'asset_due_return'
    related_entity_type VARCHAR(50),  -- 'task' or 'asset'
    related_entity_id UUID,
    message TEXT,
    is_read BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT NOW()
);

CREATE INDEX idx_notifications_user_unread ON notifications(user_id, is_read) WHERE is_read = FALSE;

5.2 Row-Level Security Enforcement
// Middleware example
const applyRowLevelSecurity = async (query, user, role, context) => {
  if (role === 'viewer' || role === 'member') {
    // Only see own department
    query.where('department_id', '=', user.department_id);
  }
  
  if (role === 'member') {
    // Can't see other members' workload
    query.exclude('workload_report');
  }
  
  return query;
};
