3.2 SQL DDL Statements
-- USERS TABLE
CREATE TABLE users (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  email VARCHAR(255) UNIQUE NOT NULL COLLATE NOCASE,
  fullName VARCHAR(100) NOT NULL,
  phoneNumber VARCHAR(20),
  avatarUrl TEXT,
  passwordHash VARCHAR(255) NOT NULL,
  roleId UUID NOT NULL REFERENCES roles(id),
  storeId UUID REFERENCES stores(id),
  status VARCHAR(20) DEFAULT 'pending_invitation' CHECK (
    status IN ('pending_invitation', 'active', 'suspended', 'deactivated')
  ),
  twoFactorEnabled BOOLEAN DEFAULT FALSE,
  twoFactorSecret VARCHAR(255),
  failedLoginAttempts INTEGER DEFAULT 0,
  lockedUntil TIMESTAMP,
  passwordChangedAt TIMESTAMP DEFAULT NOW(),
  createdAt TIMESTAMP DEFAULT NOW(),
  lastLoginAt TIMESTAMP,
  updatedAt TIMESTAMP DEFAULT NOW()
);

CREATE INDEX idx_users_email ON users(email);
CREATE INDEX idx_users_status ON users(status);
CREATE INDEX idx_users_role ON users(roleId);
CREATE INDEX idx_users_store ON users(storeId);

-- INVITATIONS TABLE
CREATE TABLE invitations (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  email VARCHAR(255) NOT NULL COLLATE NOCASE,
  tokenHash VARCHAR(255) NOT NULL,
  roleId UUID NOT NULL REFERENCES roles(id),
  storeId UUID REFERENCES stores(id),
  createdBy UUID REFERENCES users(id),
  customMessage TEXT,
  status VARCHAR(20) DEFAULT 'pending' CHECK (
    status IN ('pending', 'accepted', 'expired', 'revoked')
  ),
  expiresAt TIMESTAMP NOT NULL,
  acceptedAt TIMESTAMP,
  ipAddress INET,
  userAgent TEXT,
  createdAt TIMESTAMP DEFAULT NOW(),
  CONSTRAINT chk_expires_at CHECK (expiresAt > createdAt)
);

CREATE INDEX idx_invitations_token ON invitations(tokenHash);
CREATE INDEX idx_invitations_email ON invitations(email);
CREATE INDEX idx_invitations_status ON invitations(status);

-- ROLES TABLE
CREATE TABLE roles (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  name VARCHAR(50) UNIQUE NOT NULL,
  description TEXT,
  permissions JSONB NOT NULL DEFAULT '{}',
  isSystemRole BOOLEAN DEFAULT FALSE,
  storeScopeId UUID REFERENCES stores(id), -- Null = global scope
  createdAt TIMESTAMP DEFAULT NOW(),
  updatedAt TIMESTAMP DEFAULT NOW()
);

-- TEAMS TABLE
CREATE TABLE teams (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  name VARCHAR(100) NOT NULL,
  storeId UUID NOT NULL REFERENCES stores(id),
  leadId UUID REFERENCES users(id),
  maxMembers INTEGER DEFAULT 10,
  shiftType VARCHAR(20) DEFAULT 'mixed', -- morning, evening, mixed
  createdAt TIMESTAMP DEFAULT NOW(),
  updatedAt TIMESTAMP DEFAULT NOW()
);

CREATE INDEX idx_teams_store ON teams(storeId);

-- TEAM MEMBERSHIPS TABLE
CREATE TABLE team_memberships (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  userId UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  teamId UUID NOT NULL REFERENCES teams(id) ON DELETE CASCADE,
  roleInTeam VARCHAR(20) DEFAULT 'member' CHECK (
    roleInTeam IN ('lead', 'member')
  ),
  joinedAt TIMESTAMP DEFAULT NOW(),
  leftAt TIMESTAMP,
  createdBy UUID REFERENCES users(id),
  UNIQUE(userId, teamId)
);

CREATE INDEX idx_membership_user ON team_memberships(userId);
CREATE INDEX idx_membership_team ON team_memberships(teamId);

-- AUDIT LOGS TABLE
CREATE TABLE audit_logs (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  actorId UUID REFERENCES users(id),
  action VARCHAR(100) NOT NULL,
  entityType VARCHAR(50) NOT NULL,
  entityId UUID NOT NULL,
  oldValues JSONB,
  newValues JSONB,
  ipAddress INET,
  userAgent TEXT,
  sessionId UUID,
  createdAt TIMESTAMP DEFAULT NOW()
);

CREATE INDEX idx_audit_actor ON audit_logs(actorId);
CREATE INDEX idx_audit_entity ON audit_logs(entityType, entityId);
CREATE INDEX idx_audit_created ON audit_logs(createdAt DESC);

-- SESSIONS TABLE
CREATE TABLE sessions (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  userId UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  refreshTokenHash VARCHAR(255) NOT NULL,
  userAgent TEXT,
  ipAddress INET,
  expiresAt TIMESTAMP NOT NULL,
  revokedAt TIMESTAMP,
  createdAt TIMESTAMP DEFAULT NOW()
);

CREATE INDEX idx_sessions_user ON sessions(userId);
CREATE INDEX idx_sessions_refresh ON sessions(refreshTokenHash);

-- FRANCHISE CODES TABLE (Cho public signup)
CREATE TABLE franchise_codes (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  code VARCHAR(50) UNIQUE NOT NULL,
  storeId UUID NOT NULL REFERENCES stores(id),
  region VARCHAR(50),
  usedBy UUID REFERENCES users(id),
  usedAt TIMESTAMP,
  expiresAt TIMESTAMP,
  maxUses INTEGER DEFAULT 1, -- -1 for unlimited
  currentUses INTEGER DEFAULT 0,
  isActive BOOLEAN DEFAULT TRUE
);

6. BACKEND IMPLEMENTATION LOGIC
6.1 Middleware Chain
// Authentication middleware stack
export const authMiddleware = [
  validateJwt(),
  checkAccessTokenExpired(),
  loadUserFromToken(),
  checkAccountStatus(), // Not suspended/deactivated
  rotateRefreshToken()
];

// RBAC middleware
export const requirePermission = (...permissions) => async (req, res, next) => {
  const userPermissions = await getUserPermissions(req.user.id);
  
  const hasPermission = permissions.some(p => {
    const [resource, action] = p.split('.');
    return userPermissions[resource]?.includes(action);
  });
  
  if (!hasPermission) {
    return res.status(403).json({
      success: false,
      error: { code: 'INSUFFICIENT_PERMISSIONS' }
    });
  }
  
  next();
};

// services/invitationService.js
class InvitationService {
  async createInvitation(data) {
    const { email, roleId, storeId, createdBy, customMessage } = data;
    
    // 1. Validate email uniqueness
    const existingUser = await User.findOne({ where: { email } });
    if (existingUser && existingUser.status !== 'deactivated') {
      throw new AppError('User already exists', 409);
    }
    
    // 2. Check existing pending invitation
    const existingInvite = await Invitation.findOne({
      where: { email, status: 'pending', expiresAt: { [gt]: new Date() } }
    });
    if (existingInvite) {
      throw new AppError('Pending invitation already sent', 409);
    }
    
    // 3. Generate secure token
    const token = crypto.randomBytes(32).toString('hex');
    const tokenHash = bcrypt.hashSync(token, 10);
    const expiresAt = new Date(Date.now() + 7 * 24 * 60 * 60 * 1000); // 7 days
    
    // 4. Save to database
    const invitation = await Invitation.create({
      email,
      tokenHash,
      roleId,
      storeId,
      createdBy,
      customMessage,
      expiresAt,
      status: 'pending'
    });
    
    // 5. Send email
    await emailService.sendTemplate('invitation', {
      to: email,
      data: {
        inviteLink: `${config.frontendUrl}/auth/accept-invite?token=${token}`,
        customMessage,
        senderName: (await User.findByPk(createdBy)).fullName
      }
    });
    
    // 6. Log audit
    await AuditLog.create({
      actorId: createdBy,
      action: 'invitation_sent',
      entityType: 'invitation',
      entityId: invitation.id,
      newValues: { email, roleId, storeId }
    });
    
    return invitation;
  }
  
  async validateToken(token) {
    const tokenHash = bcrypt.hashSync(token, 10);
    
    const invitation = await Invitation.findOne({
      where: { tokenHash, status: 'pending' },
      include: [{ model: Role, attributes: ['id', 'name'] }]
    });
    
    if (!invitation) {
      throw new AppError('Invalid or expired token', 404);
    }
    
    if (invitation.expiresAt < new Date()) {
      await invitation.update({ status: 'expired' });
      throw new AppError('Invitation has expired', 410);
    }
    
    return invitation;
  }
}
6.3 Permission Engine
// services/permissionService.js
class PermissionService {
  async getUserPermissions(userId) {
    const user = await User.findByPk(userId, {
      include: [{ model: Role }]
    });
    
    if (!user) return {};
    if (user.status !== 'active') return {};
    
    let permissions = user.role.permissions || {};
    
    // Merge store-scoped roles if applicable
    if (user.storeId) {
      const storeRoles = await Role.findAll({
        where: { storeScopeId: user.storeId }
      });
      storeRoles.forEach(role => {
        permissions = this.mergePermissions(permissions, role.permissions);
      });
    }
    
    // Cache in Redis for 1 hour
    await redis.setex(`perms:${userId}`, 3600, JSON.stringify(permissions));
    
    return permissions;
  }
  
  hasPermission(userPermissions, resource, action) {
    const resourcePerms = userPermissions[resource] || [];
    
    // Check for wildcard permissions
    if (resourcePerms.includes('*')) return true;
    if (resource === '*') return true;
    
    return resourcePerms.includes(action);
  }
  
  async checkMemberAccess(requestedUserId, currentUserId, action) {
    const currentUser = await User.findByPk(currentUserId, {
      include: [{ model: Role }, { model: Store }]
    });
    
    const requestedUser = await User.findByPk(requestedUserId, {
      include: [{ model: Store }]
    });
    
    // Super Admin bypass
    if (currentUser.role.name === 'super_admin') return true;
    
    // Store Manager can manage own store
    if (currentUser.role.name === 'store_manager') {
      if (currentUser.storeId === requestedUser.storeId) {
        return ['read', 'invite'].includes(action);
      }
    }
    
    // Team Lead can manage own team members
    if (currentUser.role.name === 'team_lead') {
      const isInSameTeam = await TeamMembership.findOne({
        where: {
          userId: requestedUserId,
          teamId: { [in]: (await this.getUserTeams(currentUser.id)) }
        }
      });
      if (isInSameTeam) return true;
    }
    
    return false;
  }
}

// validators/passwordValidator.js
export const validatePassword = (password) => {
  const errors = [];
  
  // Length check
  if (password.length < 8) {
    errors.push('Password must be at least 8 characters');
  }
  if (password.length > 128) {
    errors.push('Password must be less than 128 characters');
  }
  
  // Complexity checks
  if (!/[A-Z]/.test(password)) {
    errors.push('Password must contain at least one uppercase letter');
  }
  if (!/[a-z]/.test(password)) {
    errors.push('Password must contain at least one lowercase letter');
  }
  if (!/[0-9]/.test(password)) {
    errors.push('Password must contain at least one number');
  }
  if (!/[!@#$%^&*(),.?":{}|<>]/.test(password)) {
    errors.push('Password must contain at least one special character');
  }
  
  // Blacklist common passwords
  const commonPasswords = ['password123', '123456', 'qwerty', 'admin123'];
  if (commonPasswords.includes(password.toLowerCase())) {
    errors.push('Common password not allowed');
  }
  
  // Prevent user info in password
  if (password.toLowerCase().includes(user.email?.split('@')[0])) {
    errors.push('Password cannot contain username');
  }
  
  return {
    valid: errors.length === 0,
    errors
  };
};

// middleware/bruteForceProtection.js
const loginAttempts = new Map();

export const protectBruteForce = async (req, res, next) => {
  const identifier = req.body.email || req.ip;
  const attempts = loginAttempts.get(identifier) || { count: 0, resetAt: null };
  
  // Reset after window period
  if (attempts.resetAt && attempts.resetAt < Date.now()) {
    loginAttempts.delete(identifier);
    return next();
  }
  
  // Check if locked out
  if (attempts.count >= 5) {
    return res.status(429).json({
      success: false,
      error: {
        code: 'TOO_MANY_ATTEMPTS',
        message: `Too many failed attempts. Try again in ${Math.ceil((attempts.resetAt - Date.now()) / 60000)} minutes`
      }
    });
  }
  
  // Increment counter
  attempts.count += 1;
  if (attempts.count === 3) {
    attempts.resetAt = Date.now() + 30 * 60 * 1000; // Lockout 30 minutes
    // Alert admin
    await sendAlertEmail('admin@kcoffee.com', `Suspicious login attempts from ${identifier}`);
  }
  
  loginAttempts.set(identifier, attempts);
  next();
};

import jwt from 'jsonwebtoken';
import Redis from 'ioredis';

const redis = new Redis(config.redisUrl);

export const generateTokens = (userId, rememberMe = false) => {
  const accessToken = jwt.sign(
    { userId, type: 'access' },
    config.jwtSecret,
    { expiresIn: rememberMe ? '7d' : '1h' }
  );
  
  const refreshToken = jwt.sign(
    { userId, type: 'refresh' },
    config.refreshSecret,
    { expiresIn: '30d' }
  );
  
  // Store refresh token hash in Redis
  const refreshHash = crypto.createHash('sha256').update(refreshToken).digest('hex');
  await redis.setex(
    `refresh:${refreshHash}`,
    30 * 24 * 60 * 60, // 30 days
    JSON.stringify({ userId, createdAt: Date.now(), userAgent })
  );
  
  return { accessToken, refreshToken };
};

export const rotateRefreshToken = async (req, res, next) => {
  const refreshToken = req.headers.authorization?.split(' ')[1];
  
  if (!refreshToken) return next();
  
  const refreshHash = crypto.createHash('sha256').update(refreshToken).digest('hex');
  const session = await redis.get(`refresh:${refreshHash}`);
  
  if (!session) {
    return res.status(401).json({ error: 'Invalid session' });
  }
  
  // Revoke old token
  await redis.del(`refresh:${refreshHash}`);
  
  // Generate new tokens
  const parsed = JSON.parse(session);
  const { accessToken, refreshToken: newRefreshToken } = generateTokens(parsed.userId);
  
  res.cookie('refreshToken', newRefreshToken, {
    httpOnly: true,
    secure: config.production,
    sameSite: 'strict',
    maxAge: 30 * 24 * 60 * 60 * 1000
  });
  
  next();
};

const trapFocusInModal = (modalRef) => {
  const focusableElements = modalRef.current.querySelectorAll(
    'button, [href], input, select, textarea, [tabindex]:not([tabindex="-1"])'
  );
  const firstElement = focusableElements[0];
  const lastElement = focusableElements[focusableElements.length - 1];
  
  modalRef.current.addEventListener('keydown', (e) => {
    if (e.key === 'Tab') {
      if (e.shiftKey && document.activeElement === firstElement) {
        e.preventDefault();
        lastElement.focus();
      } else if (!e.shiftKey && document.activeElement === lastElement) {
        e.preventDefault();
        firstElement.focus();
      }
    }
  });
};
