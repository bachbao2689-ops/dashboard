3.2 SQL DDL Statements
-- PROJECTS TABLE
CREATE TABLE projects (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  name VARCHAR(150) NOT NULL,
  description TEXT,
  client VARCHAR(100),
  department VARCHAR(50), -- Ecommerce, Marketing, Retail, Internal
  category VARCHAR(50), -- KV, Social Post, Banner, Packaging, etc.
  budget DECIMAL(10,2),
  actualCost DECIMAL(10,2) DEFAULT 0,
  startDate DATE NOT NULL,
  endDate DATE NOT NULL,
  priority VARCHAR(20) DEFAULT 'medium' CHECK (priority IN ('low','medium','high','urgent')),
  status VARCHAR(20) DEFAULT 'planning' CHECK (status IN ('planning','active','on_hold','completed','cancelled')),
  ownerId UUID REFERENCES users(id),
  createdAt TIMESTAMP DEFAULT NOW(),
  updatedAt TIMESTAMP DEFAULT NOW()
);

CREATE INDEX idx_projects_status ON projects(status);
CREATE INDEX idx_projects_priority ON projects(priority);
CREATE INDEX idx_projects_department ON projects(department);

-- DESIGN TASKS TABLE
CREATE TABLE design_tasks (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  taskId UUID REFERENCES tasks(id) ON DELETE CASCADE, -- Links to main task system
  projectId UUID NOT NULL REFERENCES projects(id),
  title VARCHAR(200) NOT NULL,
  brief TEXT, -- Original request brief
  requirements JSONB DEFAULT '[]', -- List of deliverables
  specifications JSONB DEFAULT '{}', -- Size, format, brand guidelines
  deliverables JSONB DEFAULT '[]', -- Expected final files
  
  -- Assignment
  designerId UUID REFERENCES users(id),
  reviewerId UUID REFERENCES users(id),
  stakeholderId UUID REFERENCES users(id), -- Who requested
  
  -- Tracking
  status VARCHAR(30) DEFAULT 'requested' CHECK (status IN (
    'requested', 'draft', 'in_progress', 'ready_for_review', 
    'revision_needed', 'approved', 'delivered', 'rejected'
  )),
  priority VARCHAR(20) DEFAULT 'medium' CHECK (priority IN ('low','medium','high','urgent')),
  
  -- Timing
  estimatedHours DECIMAL(4,2),
  actualHours DECIMAL(4,2),
  dueDate TIMESTAMP NOT NULL,
  startedAt TIMESTAMP,
  submittedForReviewAt TIMESTAMP,
  approvedAt TIMESTAMP,
  deliveredAt TIMESTAMP,
  
  -- Metrics
  revisionCount INTEGER DEFAULT 0,
  approvalRound INTEGER DEFAULT 1,
  isUrgent BOOLEAN DEFAULT FALSE,
  
  -- Metadata
  tags VARCHAR(50)[],
  createdAt TIMESTAMP DEFAULT NOW(),
  updatedAt TIMESTAMP DEFAULT NOW()
);

CREATE INDEX idx_design_tasks_project ON design_tasks(projectId);
CREATE INDEX idx_design_tasks_designer ON design_tasks(designerId);
CREATE INDEX idx_design_tasks_status ON design_tasks(status);
CREATE INDEX idx_design_tasks_due_date ON design_tasks(dueDate);

-- REVISIONS TABLE (VERSION TRACKING)
CREATE TABLE revisions (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  taskId UUID NOT NULL REFERENCES design_tasks(id) ON DELETE CASCADE,
  version INTEGER NOT NULL, -- v1, v2, v3...
  title VARCHAR(200),
  description TEXT, -- What changed in this version
  
  -- Files
  previewUrl TEXT NOT NULL, -- Web preview (webp/jpg)
  downloadUrl TEXT NOT NULL, -- Original file (PSD,AI,FIG)
  fileSize INTEGER NOT NULL, -- Bytes
  mimeType VARCHAR(100),
  metadata JSONB DEFAULT '{}', -- Dimensions, layers, etc.
  
  -- Workflow
  createdById UUID NOT NULL REFERENCES users(id),
  status VARCHAR(20) DEFAULT 'submitted' CHECK (status IN ('submitted','reviewed','approved','rejected')),
  feedback TEXT, -- Reviewer comments
  approvedById UUID REFERENCES users(id),
  rejectedReason TEXT,
  
  -- Timestamps
  createdAt TIMESTAMP DEFAULT NOW(),
  reviewedAt TIMESTAMP,
  
  UNIQUE(taskId, version)
);

CREATE INDEX idx_revisions_task ON revisions(taskId);
CREATE INDEX idx_reversions_version ON revisions(taskId, version DESC);

-- DESIGN ASSETS LIBRARY
CREATE TABLE design_assets (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  name VARCHAR(200) NOT NULL,
  description TEXT,
  category VARCHAR(50), -- Templates, Logos, Icons, Fonts, Colors
  subcategory VARCHAR(50),
  fileType VARCHAR(20), -- PSD, AI, FIGMA, SVG, PNG, PDF
  fileUrl TEXT NOT NULL,
  thumbnailUrl TEXT,
  
  -- Brand metadata
  brandId UUID REFERENCES brands(id),
  tags VARCHAR(50)[],
  usageRights VARCHAR(50), -- Internal, Commercial, Licensed
  lastUpdated TIMESTAMP,
  uploadedBy UUID REFERENCES users(id),
  downloadCount INTEGER DEFAULT 0,
  
  createdAt TIMESTAMP DEFAULT NOW()
);

CREATE INDEX idx_assets_category ON design_assets(category);
CREATE INDEX idx_assets_tags ON design_assets USING GIN(tags);

-- BRAND GUIDELINES
CREATE TABLE brand_guidelines (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  brandName VARCHAR(100) NOT NULL,
  version VARCHAR(20),
  logoVariations JSONB DEFAULT '[]',
  colorPalette JSONB DEFAULT '[]', -- {name, hex, usage}
  typography JSONB DEFAULT '[]', -- Fonts, sizes, hierarchy
  templates JSONB DEFAULT '[]', -- Downloadable templates
  doAndDonots JSONB DEFAULT '{}',
  pdfUrl TEXT,
  isActive BOOLEAN DEFAULT TRUE,
  updatedAt TIMESTAMP DEFAULT NOW()
);

-- COMMENTS ON DESIGNS (PINNED ANNOTATIONS)
CREATE TABLE design_comments (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  revisionId UUID NOT NULL REFERENCES revisions(id) ON DELETE CASCADE,
  userId UUID NOT NULL REFERENCES users(id),
  xCoordinate INTEGER NOT NULL, -- Position on image
  yCoordinate INTEGER NOT NULL,
  width INTEGER, -- Bubble size
  height INTEGER,
  content TEXT NOT NULL,
  resolved BOOLEAN DEFAULT FALSE,
  resolvedById UUID REFERENCES users(id),
  resolvedAt TIMESTAMP,
  parentId UUID REFERENCES design_comments(id), -- For replies
  createdAt TIMESTAMP DEFAULT NOW()
);

CREATE INDEX idx_comments_revision ON design_comments(revisionId);
CREATE INDEX idx_comments_resolved ON design_comments(resolved);

6.1 Revision State Machine
// Revision lifecycle
const revisionStates = {
  submitted: {
    nextAllowed: ['reviewed', 'approved', 'rejected'],
    triggers: ['designer uploads file']
  },
  reviewed: {
    nextAllowed: ['approved', 'rejected'],
    triggers: ['reviewer gives feedback']
  },
  approved: {
    nextAllowed: ['final_approved'],
    triggers: ['manager approves']
  },
  rejected: {
    nextAllowed: ['submitted'], // Need new version
    triggers: ['designer submits v(n+1)']
  }
};

// Max revision rule
const MAX_REVISIONS = 5;
function checkRevisionLimit(currentVersion) {
  if (currentVersion >= MAX_REVISIONS) {
    return {
      allowed: false,
      message: `Maximum ${MAX_REVISIONS} revisions reached. Escalate to manager.`
    };
  }
  return { allowed: true };
}
6.2 Auto-Assignment Algorithm
// Assign task to least loaded designer
function autoAssignDesigner(projectId, requiredSkills) {
  const designers = await getDesignersByProject(projectId);
  
  const availabilityScores = await Promise.all(
    designers.map(async (designer) => {
      const workload = await getDesignerWorkload(designer.id);
      const skillMatch = calculateSkillMatch(designer.skills, requiredSkills);
      
      return {
        designer,
        score: {
          availability: (workload.availableHours / workload.capacity) * 0.7,
          skillMatch: skillMatch * 0.3
        },
        totalScore: ((workload.availableHours / workload.capacity) * 0.7) + (skillMatch * 0.3)
      };
    })
  );
  
  // Sort by total score descending
  availabilityScores.sort((a, b) => b.totalScore - a.totalScore);
  
  return availabilityScores[0].designer;
}
6.3 Deadline Alert Logic
// Calculate urgency and send alerts
function checkTaskDeadlines() {
  const today = new Date();
  
  tasks.forEach(task => {
    const daysRemaining = Math.ceil((task.dueDate - today) / (1000 * 60 * 60 * 24));
    
    if (task.status === 'in_progress' && daysRemaining <= 2) {
      // Send urgent alert
      sendNotification({
        type: 'deadline_urgent',
        to: [task.designerId, task.reviewerId],
        message: `Task "${task.title}" due in ${daysRemaining} days`,
        priority: 'high'
      });
    }
    
    if (task.status === 'in_progress' && daysRemaining <= 0) {
      // Already overdue
      task.isUrgent = true;
      notifyManager(task.id, 'Task overdue - escalation needed');
    }
    
    if (task.revisionCount >= 3 && daysRemaining <= 3) {
      // High revision count with tight deadline
      warnAboutQualityRisk(task.id);
    }
  });
}

// Run cron job every hour
scheduleCron('0 * * * *', checkTaskDeadlines);

// Auto-generate web preview from source
async function generatePreview(sourceFile) {
  const ext = getFileExtension(sourceFile.name);
  
  if (supportedFormats.source.includes(ext)) {
    // Use ImageMagick/GraphicsMagick for raster
    // Use Sharp for conversion to WebP
    const previewBuffer = await sharp(sourceFile.buffer)
      .resize(800, 600, { fit: 'inside' })
      .webp({ quality: 80 })
      .toBuffer();
    
    return {
      previewUrl: `https://cdn.kcoffee.com/previews/${uuid}.webp`,
      thumbnailUrl: `https://cdn.kcoffee.com/thumbs/${uuid}.jpg`
    };
  }
}
