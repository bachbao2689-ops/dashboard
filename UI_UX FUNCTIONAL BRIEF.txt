const availabilityCheck = {
  trigger: 'onBlur',
  fields: ['asset', 'borrowDate', 'dueDate'],
  endpoint: '/api/v1/assets/check-availability',
  onError: (conflict) => {
    showWarning('Asset is already booked during these dates');
    suggestAlternativeDates(conflict.availableSlots);
  }
};
